Basketball Hoop Trigger

Creating a trigger collider to detect successful shots

Setting Up the Hoop
  1. Create a basketball hoop with backboard, rim, and net using primitive objects or imported models
  2. Add a Box Collider to the backboard and rim with Is Trigger = false (solid objects)
  3. Create an empty GameObject as a child of the rim and position it below the rim
  4. Add a Box Collider to this GameObject and set Is Trigger = true
  5. Size the trigger collider to detect when the ball passes through the hoop

Implementing Score Detection 
using UnityEngine;

public class HoopTrigger : MonoBehaviour
{
    // Reference to the score manager
    public ScoreManager scoreManager;
    
    // Flag to prevent multiple triggers from one shot
    private bool canScore = true;
    
    // Audio for successful shot
    public AudioSource scoreSound;
    
    private void OnTriggerEnter(Collider other)
    {
        // Check if the basketball entered the trigger
        if (other.CompareTag("Basketball") && canScore)
        {
            // Increment score
            scoreManager.AddPoints(2);
            
            // Play sound effect
            if (scoreSound != null)
            {
                scoreSound.Play();
            }
            
            // Prevent multiple triggers
            canScore = false;
            Invoke("ResetScoring", 1.0f);
        }
    }
    
    private void ResetScoring()
    {
        canScore = true;
    }
}
