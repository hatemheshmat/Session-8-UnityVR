Enhancing gameplay with audio feedback

Setting Up Audio Sources
  1. Import audio files into your project (WAV or MP3 format)
  2. Create an empty GameObject named "AudioManager"
  3. Add multiple AudioSource components to the AudioManager
  4. Assign different sound effects to each AudioSource
  5. Configure settings: Spatial Blend (3D sound), Volume, Pitch

Audio Manager Script 
using UnityEngine;

public class AudioManager : MonoBehaviour
{
    // Singleton instance
    public static AudioManager Instance;
    
    // Audio sources for different sound effects
    public AudioSource swishSound;
    public AudioSource bounceSound;
    public AudioSource cheerSound;
    public AudioSource buzzerSound;
    
    // Volume control
    public float masterVolume = 1.0f;
    private float originalSwishVolume;
    
    void Awake()
    {
        // Singleton pattern
        if (Instance == null)
        {
            Instance = this;
            DontDestroyOnLoad(gameObject);
            
            // Store original volumes
            if (swishSound != null)
                originalSwishVolume = swishSound.volume;
        }
        else if (Instance != this)
        {
            Destroy(gameObject);
        }
    }
    
    public void PlaySwish()
    {
        if (swishSound != null && !swishSound.isPlaying)
        {
            swishSound.Play();
        }
    }
    
    public void PlayBounce(float intensity = 1.0f)
    {
        if (bounceSound != null)
        {
            // Adjust pitch based on bounce intensity
            bounceSound.pitch = Mathf.Lerp(0.8f, 1.2f, intensity);
            bounceSound.Play();
        }
    }
    
    public void PlayCheer()
    {
        if (cheerSound != null)
        {
            cheerSound.Play();
        }
    }
    
    public void PlayBuzzer()
    {
        if (buzzerSound != null)
        {
            buzzerSound.Play();
        }
    }
    
    public void SetMasterVolume(float volume)
    {
        masterVolume = Mathf.Clamp01(volume);
        
        // Apply to all audio sources
        if (swishSound != null)
            swishSound.volume = originalSwishVolume * masterVolume;
            
        // Apply to other sounds similarly
    }
}
// In HoopTrigger.cs
private void OnTriggerEnter(Collider other)
{
    if (other.CompareTag("Basketball") && canScore)
    {
        // Increment score
        scoreManager.AddPoints(2);
        
        // Play sound effect
        AudioManager.Instance.PlaySwish();
        
        // Prevent multiple triggers
        canScore = false;
        Invoke("ResetScoring", 1.0f);
    }
}
