Creating the Game Board

Building an interactive Tic-Tac-Toe board in VR

Board Setup Steps
  1. Create a parent GameObject named "TicTacToeBoard"
  2. Add a flat Cube as the board base and scale it appropriately
  3. Create a 3×3 grid of smaller Cubes as cell placeholders
  4. Add Box Colliders to each cell with Is Trigger = true
  5. Create X and O prefabs with distinctive materials
  6. Add the TicTacToeManager script to the board parent

VR Interaction Components
XR Ray Interactor
Allows players to point at cells from a distance
XR Interactable
Makes each cell responsive to controller interaction
Interaction Manager
Coordinates between interactors and interactables
using UnityEngine;
using UnityEngine.XR.Interaction.Toolkit;

public class TicTacToeCell : MonoBehaviour
{
    // Reference to the game manager
    public TicTacToeManager gameManager;
    
    // Cell position in the grid
    public int row;
    public int column;
    
    // Reference to XR components
    private XRSimpleInteractable interactable;
    
    // References for visual feedback
    public Material defaultMaterial;
    public Material hoverMaterial;
    private MeshRenderer meshRenderer;
    
    void Start()
    {
        // Get components
        interactable = GetComponent<XRSimpleInteractable>();
        meshRenderer = GetComponent<MeshRenderer>();
        
        // Set up interaction events
        interactable.selectEntered.AddListener(OnCellSelected);
        interactable.hoverEntered.AddListener(OnCellHoverEnter);
        interactable.hoverExited.AddListener(OnCellHoverExit);
    }
    
    private void OnCellSelected(SelectEnterEventArgs args)
    {
        // Tell the game manager a move was made
        gameManager.MakeMove(row, column);
    }
    
    private void OnCellHoverEnter(HoverEnterEventArgs args)
    {
        // Visual feedback when hovering
        meshRenderer.material = hoverMaterial;
    }
    
    private void OnCellHoverExit(HoverExitEventArgs args)
    {
        // Reset material when not hovering
        meshRenderer.material = defaultMaterial;
    }
}
