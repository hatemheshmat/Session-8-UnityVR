Ball Reset Mechanism

Implementing continuous gameplay with ball respawning

Reset Methods
Time-Based Reset
Reset ball after a set time period
Pros & Cons:
  • Simple to implement
  • Works regardless of ball location
  • May reset during active play
Zone-Based Reset
Reset when ball enters a specific area
Pros & Cons:
  • More natural gameplay flow
  • Player has control over reset
  • Ball may get stuck out of reach
Reset Zone
Implementation
  1. Create a reset zone with a trigger collider
  2. Define a spawn point for the basketball
  3. Implement a backup time-based reset (failsafe)
  4. Add visual/audio feedback for reset events 
using UnityEngine;

public class BallResetManager : MonoBehaviour
{
    // Reference to the basketball
    public GameObject basketball;
    
    // Spawn point transform
    public Transform spawnPoint;
    
    // Time-based reset parameters
    public float maxBallLifetime = 15.0f;
    private float ballTimer = 0f;
    
    // Reference to basketball's rigidbody
    private Rigidbody basketballRb;
    
    void Start()
    {
        if (basketball != null)
        {
            basketballRb = basketball.GetComponent<Rigidbody>();
        }
    }
    
    void Update()
    {
        // Time-based failsafe reset
        if (basketball != null)
        {
            ballTimer += Time.deltaTime;
            
            if (ballTimer >= maxBallLifetime)
            {
                ResetBall();
            }
        }
    }
    
    public void ResetBall()
    {
        if (basketball != null && spawnPoint != null)
        {
            // Reset position and rotation
            basketball.transform.position = spawnPoint.position;
            basketball.transform.rotation = spawnPoint.rotation;
            
            // Reset physics
            if (basketballRb != null)
            {
                basketballRb.velocity = Vector3.zero;
                basketballRb.angularVelocity = Vector3.zero;
            }
            
            // Reset timer
            ballTimer = 0f;
            
            // Optional: Play reset sound
            AudioManager.Instance?.PlayResetSound();
        }
    }
}

// For the reset zone:
public class ResetZoneTrigger : MonoBehaviour
{
    public BallResetManager resetManager;
    
    private void OnTriggerEnter(Collider other)
    {
        if (other.CompareTag("Basketball") && resetManager != null)
        {
            resetManager.ResetBall();
        }
    }
}
