Calculating Throw Velocity

Improving throwing mechanics with controller velocity tracking

BasketballController.cs 
using UnityEngine;
using UnityEngine.XR.Interaction.Toolkit;

public class BasketballController : MonoBehaviour
{
    private Rigidbody rb;
    private XRGrabInteractable grabInteractable;
    
    // Velocity tracking variables
    private Vector3 previousPosition;
    private Vector3[] velocityFrames = new Vector3[5];
    private int frameIndex = 0;
    private float throwMultiplier = 1.5f;
    
    void Awake()
    {
        rb = GetComponent<Rigidbody>();
        grabInteractable = GetComponent<XRGrabInteractable>();
        
        // Subscribe to interaction events
        grabInteractable.selectEntered.AddListener(OnGrab);
        grabInteractable.selectExited.AddListener(OnRelease);
    }
    
    void OnGrab(SelectEnterEventArgs args)
    {
        // Reset velocity tracking when grabbed
        for (int i = 0; i < velocityFrames.Length; i++)
        {
            velocityFrames[i] = Vector3.zero;
        }
        
        // Start tracking position
        previousPosition = transform.position;
    }
    
    void Update()
    {
        if (grabInteractable.isSelected)
        {
            // Track velocity while held
            Vector3 currentVelocity = (transform.position - previousPosition) / Time.deltaTime;
            velocityFrames[frameIndex] = currentVelocity;
            frameIndex = (frameIndex + 1) % velocityFrames.Length;
            previousPosition = transform.position;
        }
    }
    
    void OnRelease(SelectExitEventArgs args)
    {
        // Calculate average velocity from stored frames
        Vector3 averageVelocity = Vector3.zero;
        
        foreach (Vector3 velocity in velocityFrames)
        {
            averageVelocity += velocity;
        }
        
        averageVelocity /= velocityFrames.Length;
        
        // Apply calculated velocity with multiplier
        rb.velocity = averageVelocity * throwMultiplier;
    }
}
Why Track Velocity?

The default XR Grab Interactable component doesn't always provide realistic throwing physics. By tracking the controller's movement over multiple frames, we can calculate a more accurate velocity vector that matches the user's throwing motion.

This creates a more natural and intuitive throwing experience, especially important for a basketball game where accurate throws are essential.

How It Works

1. We store velocity samples across multiple frames in an array

2. On each Update(), we calculate the current velocity and store it

3. When the ball is released, we average all stored velocities

4. We apply this averaged velocity to the Rigidbody with a multiplier
