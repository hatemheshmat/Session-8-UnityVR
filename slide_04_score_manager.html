Score Manager & UI Display

Tracking and displaying the player's score in VR

Score Manager Implementation 
using UnityEngine;
using TMPro;

public class ScoreManager : MonoBehaviour
{
    // Current score value
    private int currentScore = 0;
    
    // Reference to TextMeshPro text component
    public TextMeshProUGUI scoreText;
    
    // Optional high score tracking
    private int highScore = 0;
    public TextMeshProUGUI highScoreText;
    
    // Animation reference for score change
    public Animator scoreAnimator;
    
    void Start()
    {
        // Initialize score display
        UpdateScoreDisplay();
        
        // Load high score from PlayerPrefs
        highScore = PlayerPrefs.GetInt("HighScore", 0);
        if (highScoreText != null)
        {
            highScoreText.text = "High: " + highScore.ToString();
        }
    }
    
    public void AddPoints(int points)
    {
        // Increment score
        currentScore += points;
        
        // Update UI
        UpdateScoreDisplay();
        
        // Trigger animation if available
        if (scoreAnimator != null)
        {
            scoreAnimator.SetTrigger("ScoreChanged");
        }
        
        // Check for high score
        if (currentScore > highScore)
        {
            highScore = currentScore;
            PlayerPrefs.SetInt("HighScore", highScore);
            
            if (highScoreText != null)
            {
                highScoreText.text = "High: " + highScore.ToString();
            }
        }
    }
    
    private void UpdateScoreDisplay()
    {
        if (scoreText != null)
        {
            scoreText.text = "Score: " + currentScore.ToString();
        }
    }
    
    public void ResetScore()
    {
        currentScore = 0;
        UpdateScoreDisplay();
    }
}
Setting Up the UI Canvas
  1. Create a new World Space Canvas in your scene
  2. Position the canvas where it's easily visible to the player (e.g., near the hoop)
  3. Add a Panel component as a background for better visibility
  4. Add TextMeshPro - Text (UI) components for score and high score
  5. Create a ScoreManager GameObject and attach the ScoreManager script
  6. Assign the TextMeshPro components to the ScoreManager in the Inspector
