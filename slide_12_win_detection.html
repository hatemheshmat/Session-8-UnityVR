public bool CheckWin(int player)
{
    // Check rows
    for (int row = 0; row < 3; row++)
    {
        if (board[row, 0] == player && 
            board[row, 1] == player && 
            board[row, 2] == player)
        {
            // Highlight winning cells
            HighlightWinningCells(row, 0, row, 1, row, 2);
            return true;
        }
    }
    
    // Check columns
    for (int col = 0; col < 3; col++)
    {
        if (board[0, col] == player && 
            board[1, col] == player && 
            board[2, col] == player)
        {
            // Highlight winning cells
            HighlightWinningCells(0, col, 1, col, 2, col);
            return true;
        }
    }
    
    // Check diagonal (top-left to bottom-right)
    if (board[0, 0] == player && 
        board[1, 1] == player && 
        board[2, 2] == player)
    {
        // Highlight winning cells
        HighlightWinningCells(0, 0, 1, 1, 2, 2);
        return true;
    }
    
    // Check diagonal (top-right to bottom-left)
    if (board[0, 2] == player && 
        board[1, 1] == player && 
        board[2, 0] == player)
    {
        // Highlight winning cells
        HighlightWinningCells(0, 2, 1, 1, 2, 0);
        return true;
    }
    
    return false;
}

public bool IsBoardFull()
{
    // Check if all cells are filled
    for (int row = 0; row < 3; row++)
    {
        for (int col = 0; col < 3; col++)
        {
            if (board[row, col] == 0) // Empty cell
            {
                return false;
            }
        }
    }
    return true;
}

private void HighlightWinningCells(int r1, int c1, int r2, int c2, int r3, int c3)
{
    // Get cell GameObjects from the grid
    GameObject cell1 = cellObjects[r1, c1];
    GameObject cell2 = cellObjects[r2, c2];
    GameObject cell3 = cellObjects[r3, c3];
    
    // Apply winning effect (e.g., change material)
    ApplyWinningEffect(cell1);
    ApplyWinningEffect(cell2);
    ApplyWinningEffect(cell3);
    
    // Draw a line through winning cells
    DrawWinLine(cell1.transform.position, cell3.transform.position);
}
