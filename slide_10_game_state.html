Game State Implementation

Using an enum to define possible game states makes our code more readable and maintainable.

using UnityEngine;
using TMPro;

public class TicTacToeManager : MonoBehaviour
{
    // Game state enum
    public enum GameState
    {
        Initialize,
        PlayerXTurn,
        PlayerOTurn,
        PlayerXWin,
        PlayerOWin,
        Draw
    }
    
    // Current game state
    private GameState currentState;
    
    // Game board (0=empty, 1=X, 2=O)
    private int[,] board = new int[3, 3];
    
    // UI references
    public TextMeshProUGUI statusText;
    
    void Start()
    {
        // Initialize the game
        InitializeGame();
    }
    
    void InitializeGame()
    {
        // Clear the board
        for (int i = 0; i < 3; i++)
        {
            for (int j = 0; j < 3; j++)
            {
                board[i, j] = 0;
            }
        }
        
        // Set initial state
        SetGameState(GameState.PlayerXTurn);
    }
    
    void SetGameState(GameState newState)
    {
        currentState = newState;
        
        // Update UI based on state
        switch (currentState)
        {
            case GameState.PlayerXTurn:
                statusText.text = "Player X's Turn";
                break;
            case GameState.PlayerOTurn:
                statusText.text = "Player O's Turn";
                break;
            case GameState.PlayerXWin:
                statusText.text = "Player X Wins!";
                break;
            case GameState.PlayerOWin:
                statusText.text = "Player O Wins!";
                break;
            case GameState.Draw:
                statusText.text = "Game Draw!";
                break;
        }
    }
}

public void MakeMove(int row, int col)
{
    // Check if the cell is empty and it's a valid turn
    if (board[row, col] == 0 && 
        (currentState == GameState.PlayerXTurn || 
         currentState == GameState.PlayerOTurn))
    {
        // Place X or O based on current turn
        if (currentState == GameState.PlayerXTurn)
        {
            board[row, col] = 1; // X = 1
            
            // Check for win or draw
            if (CheckWin(1))
            {
                SetGameState(GameState.PlayerXWin);
            }
            else if (IsBoardFull())
            {
                SetGameState(GameState.Draw);
            }
            else
            {
                // Switch to next player
                SetGameState(GameState.PlayerOTurn);
            }
        }
        else // PlayerOTurn
        {
            board[row, col] = 2; // O = 2
            
            // Check for win or draw
            if (CheckWin(2))
            {
                SetGameState(GameState.PlayerOWin);
            }
            else if (IsBoardFull())
            {
                SetGameState(GameState.Draw);
            }
            else
            {
                // Switch to next player
                SetGameState(GameState.PlayerXTurn);
            }
        }
        
        // Update visual representation
        UpdateBoardVisuals(row, col);
    }
}
