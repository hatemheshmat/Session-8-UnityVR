UI Setup Steps

  2. Position the canvas above or next to the game board
  3. Add a Panel component as a background
  4. Add TextMeshPro components for game status and player turn
  5. Add a Button component for the "New Game" functionality
  6. Create visual indicators for current player's turn

using UnityEngine;
using TMPro;
using UnityEngine.UI;

public class TicTacToeUI : MonoBehaviour
{
    // Reference to game manager
    public TicTacToeManager gameManager;
    
    // UI elements
    public TextMeshProUGUI statusText;
    public Button newGameButton;
    
    // Player turn indicators
    public GameObject playerXIndicator;
    public GameObject playerOIndicator;
    
    // Win/draw animations
    public Animator winAnimator;
    public Animator drawAnimator;
    
    void Start()
    {
        // Set up button listener
        newGameButton.onClick.AddListener(OnNewGameClicked);
        
        // Initial UI state
        UpdateUI(TicTacToeManager.GameState.PlayerXTurn);
    }
    
    public void UpdateUI(TicTacToeManager.GameState state)
    {
        // Update status text based on game state
        switch (state)
        {
            case TicTacToeManager.GameState.PlayerXTurn:
                statusText.text = "Player X's Turn";
                playerXIndicator.SetActive(true);
                playerOIndicator.SetActive(false);
                break;
                
            case TicTacToeManager.GameState.PlayerOTurn:
                statusText.text = "Player O's Turn";
                playerXIndicator.SetActive(false);
                playerOIndicator.SetActive(true);
                break;
                
            case TicTacToeManager.GameState.PlayerXWin:
                statusText.text = "Player X Wins!";
                playerXIndicator.SetActive(true);
                playerOIndicator.SetActive(false);
                winAnimator.SetTrigger("Win");
                break;
                
            case TicTacToeManager.GameState.PlayerOWin:
                statusText.text = "Player O Wins!";
                playerXIndicator.SetActive(false);
                playerOIndicator.SetActive(true);
                winAnimator.SetTrigger("Win");
                break;
                
            case TicTacToeManager.GameState.Draw:
                statusText.text = "Game Draw!";
                playerXIndicator.SetActive(false);
                playerOIndicator.SetActive(false);
                drawAnimator.SetTrigger("Draw");
                break;
        }
    }
    
    private void OnNewGameClicked()
    {
        // Tell the game manager to start a new game
        gameManager.StartNewGame();
    }
}
// In TicTacToeManager.cs
private TicTacToeUI uiManager;

void Start()
{
    // Get UI manager reference
    uiManager = GetComponent<TicTacToeUI>();
    
    // Initialize the game
    InitializeGame();
}

void SetGameState(GameState newState)
{
    currentState = newState;
    
    // Update UI
    if (uiManager != null)
    {
        uiManager.UpdateUI(currentState);
    }
    
    // Additional state-specific logic
    if (currentState == GameState.PlayerXWin || 
        currentState == GameState.PlayerOWin || 
        currentState == GameState.Draw)
    {
        // Disable further interaction with the board
        DisableBoardInteraction();
    }
}
