public class TicTacToeManager : MonoBehaviour
{
    // Game board (0=empty, 1=X, 2=O)
    private int[,] board = new int[3, 3];
    
    // References to X and O objects on the board
    private List<GameObject> placedPieces = new List<GameObject>();
    
    // References to cell objects
    private GameObject[,] cellObjects = new GameObject[3, 3];
    
    // Prefabs for X and O
    public GameObject xPrefab;
    public GameObject oPrefab;
    
    // Game state
    private GameState currentState;
    
    // Stats tracking
    private int xWins = 0;
    private int oWins = 0;
    private int draws = 0;
    
    public void StartNewGame()
    {
        // Clear the board array
        for (int row = 0; row < 3; row++)
        {
            for (int col = 0; col < 3; col++)
            {
                board[row, col] = 0;
            }
        }
        
        // Remove all placed X and O objects
        foreach (GameObject piece in placedPieces)
        {
            Destroy(piece);
        }
        placedPieces.Clear();
        
        // Re-enable cell interactions
        EnableBoardInteraction();
        
        // Reset any visual effects
        ResetVisualEffects();
        
        // Set initial game state
        SetGameState(GameState.PlayerXTurn);
        
        // Play reset sound
        PlayResetSound();
    }
    
    private void EnableBoardInteraction()
    {
        // Enable all cell interactables
        for (int row = 0; row < 3; row++)
        {
            for (int col = 0; col < 3; col++)
            {
                if (cellObjects[row, col] != null)
                {
                    // Get the interactable component
                    XRSimpleInteractable interactable = 
                        cellObjects[row, col].GetComponent<XRSimpleInteractable>();
                    
                    if (interactable != null)
                    {
                        interactable.enabled = true;
                    }
                }
            }
        }
    }
    
    private void DisableBoardInteraction()
    {
        // Disable all cell interactables
        for (int row = 0; row < 3; row++)
        {
            for (int col = 0; col < 3; col++)
            {
                if (cellObjects[row, col] != null)
                {
                    // Get the interactable component
                    XRSimpleInteractable interactable = 
                        cellObjects[row, col].GetComponent<XRSimpleInteractable>();
                    
                    if (interactable != null)
                    {
                        interactable.enabled = false;
                    }
                }
            }
        }
    }
    
    private void ResetVisualEffects()
    {
        // Reset any highlighting or animations
        foreach (GameObject cell in cellObjects)
        {
            if (cell != null)
            {
                // Reset material to default
                MeshRenderer renderer = cell.GetComponent<MeshRenderer>();
                if (renderer != null && defaultMaterial != null)
                {
                    renderer.material = defaultMaterial;
                }
            }
        }
    }
}

public void UpdateStats(int xWins, int oWins, int draws)
{
    // Update stats display
    if (statsText != null)
    {
        statsText.text = $"X Wins: {xWins} | O Wins: {oWins} | Draws: {draws}";
    }
}
